export class ExtractorTimeoutError extends Error {
  constructor() {
    super("Extractor process timed out");
  }
}

/** Bun's Windows fs.rm does not reliably honor Node's maxRetries option. */
export async function removeExtractorStaging(directory: string, ownerRoot: string): Promise<void> {
  const base = await realpath(ownerRoot);
  const target = resolve(directory);
  if (!contained(base, target) || target === base)
    throw new Error("Extractor cleanup target is outside its owned staging root");
  for (let attempt = 0; attempt <= 8; attempt++) {
    try {
      let info: Awaited<ReturnType<typeof lstat>>;
      try {
        info = await lstat(target);
      } catch (error: unknown) {
        if (error instanceof Error && "code" in error && error.code === "ENOENT") return;
        throw error;
      }
      if (!info.isDirectory() || info.isSymbolicLink() || !contained(base, await realpath(target)))
        throw new Error("Extractor cleanup target was replaced");
      await rm(target, { recursive: true, force: true });
      return;
    } catch (error: unknown) {
      if (
        attempt === 8 ||
        !(error instanceof Error) ||
        !("code" in error) ||
        !["EBUSY", "EPERM", "ENOTEMPTY"].includes(String(error.code))
      )
        throw error;
      await delay(100 * (attempt + 1));
    }
  }
}

/** Own the process tree until close, including cancellation; this is not an OS sandbox. */
function runExtractorProcess(
  executable: string,
  args: readonly string[],
  options: {
    signal: AbortSignal;
    cwd: string;
    env: NodeJS.ProcessEnv;
    bytes: number;
    timeoutMs: number;
  },
): Promise<Uint8Array> {
  return new Promise((resolveProcess, reject) => {
    options.signal.throwIfAborted();
    const child = spawn(executable, [...args], {
      cwd: options.cwd,
      env: options.env,
      windowsHide: true,
      shell: false,
      // A separate group also prevents Windows extractors' console control events
      // from reaching the caller. Keep the hidden process referenced until close.
      detached: true,
      stdio: ["ignore", "pipe", "pipe"],
    });
    const chunks: Buffer[] = [];
    let stdoutBytes = 0;
    let stderrBytes = 0;
    let failed = false;
    let timedOut = false;
    let termination: Promise<void> | undefined;
    const terminate = () => {
      failed = true;
      if (termination !== undefined) return;
      const pid = child.pid;
      termination = new Promise<void>((finished) => {
        if (pid === undefined) {
          finished();
          return;
        }
        if (process.platform === "win32") {
          const killer = join(process.env.SystemRoot ?? "C:\\Windows", "System32", "taskkill.exe");
          execFile(
            killer,
            ["/PID", String(pid), "/T", "/F"],
            {
              windowsHide: true,
              timeout: 10_000,
              maxBuffer: 16_384,
              env: processEnvironment(),
            },
            () => {
              child.kill("SIGKILL");
              finished();
            },
          );
        } else {
          try {
            process.kill(-pid, "SIGKILL");
          } catch {
            child.kill("SIGKILL");
          }
          finished();
        }
      });
    };
    const timer = setTimeout(() => {
      timedOut = true;
      terminate();
    }, options.timeoutMs);
    options.signal.addEventListener("abort", terminate, { once: true });
    child.stdout.on("data", (chunk: Buffer) => {
      stdoutBytes += chunk.byteLength;
      if (stdoutBytes > options.bytes) terminate();
      else chunks.push(chunk);
    });
    child.stderr.on("data", (chunk: Buffer) => {
      stderrBytes += chunk.byteLength;
      if (stderrBytes > options.bytes) terminate();
    });
    child.on("error", () => {
      failed = true;
    });
    child.on("close", (code) => {
      clearTimeout(timer);
      options.signal.removeEventListener("abort", terminate);
      void Promise.resolve(termination).then(() => {
        if (timedOut) reject(new ExtractorTimeoutError());
        else if (failed || code !== 0 || options.signal.aborted)
          reject(new Error("Extractor process failed, was cancelled, or exceeded its limit"));
        else resolveProcess(Buffer.concat(chunks, stdoutBytes));
      });
    });
    if (options.signal.aborted) terminate();
  });
}

function validateProfile(profile: ExtractorProfile): void {
  if (profile.engine === "builtin") return;
  if (
    !profile.args.some((arg) => arg.includes("{input}")) ||
    (profile.outputMode !== "stdout" && !profile.args.some((arg) => arg.includes("{outputDir}")))
  )
    throw new Error("Extractor arguments must include input and output directory placeholders");
  for (const arg of [...profile.args, ...(profile.versionArgs ?? [])]) {
    if (
      /\{[a-zA-Z][a-zA-Z0-9_]*\}/.test(arg.replaceAll("{input}", "").replaceAll("{outputDir}", ""))
    )
      throw new Error("Unknown extractor argument placeholder");
  }
  if (profile.versionArgs?.some((arg) => /\{(?:input|outputDir)\}/.test(arg)))
    throw new Error("Version probes cannot use document placeholders");
  if (profile.versionCommand !== undefined && profile.versionArgs === undefined)
    throw new Error("Version command requires version arguments");
  if (
    profile.outputMarkdown !== undefined &&
    (!safeRelative(profile.outputMarkdown) || !/\.(md|markdown)$/i.test(profile.outputMarkdown))
  )
    throw new Error("Extractor Markdown output must be a contained relative Markdown path");
  const environment = Object.entries(profile.env ?? {});
  if (
    environment.length > 64 ||
    environment.some(([key]) => !/^[A-Za-z_][A-Za-z0-9_]{0,127}$/.test(key))
  )
    throw new Error("Invalid extractor environment");
}

/** Explicit trusted local code only. No ambient, remote, dotenv or environment layers. */
export const loadExtractorConfig = Effect.fn("loadExtractorConfig")(function* (path: string) {
  const configFile = resolve(path);
  const config = yield* Effect.tryPromise({
    try: async () => {
      const info = await stat(configFile);
      if (!info.isFile() || info.size > 262_144)
        throw new Error("Invalid extractor configuration file");
      const loaded = await loadConfig<Record<string, unknown>>({
        name: "teamagent-extractors",
        cwd: dirname(configFile),
        configFile,
        configFileRequired: true,
        rcFile: false,
        globalRc: false,
        packageJson: false,
        dotenv: false,
        envName: false,
        extend: false,
        giget: false,
        omit$Keys: false,
        merger: (...sources) =>
          sources
            .toReversed()
            .reduce<Record<string, unknown>>((merged, source) => ({ ...merged, ...source }), {}),
      });
      const dynamic: unknown = loaded.config;
      const parsed = Schema.decodeUnknownSync(ExtractorConfigSchema)(dynamic, {
        onExcessProperty: "error",
      });
      const ids = new Set(parsed.profiles.map((profile) => profile.id));
      if (ids.size !== parsed.profiles.length || !ids.has(parsed.defaultProfile))
        throw new Error("Duplicate or missing extractor profile");
      const profiles = parsed.profiles.map((profile) => {
        validateProfile(profile);
        if (profile.engine === "builtin") return profile;
        const command =
          isAbsolute(profile.command) || !/[\\/]/.test(profile.command)
            ? profile.command
            : resolve(dirname(configFile), profile.command);
        const versionCommand =
          profile.versionCommand === undefined
            ? undefined
            : isAbsolute(profile.versionCommand) || !/[\\/]/.test(profile.versionCommand)
              ? profile.versionCommand
              : resolve(dirname(configFile), profile.versionCommand);
        return { ...profile, command, ...(versionCommand === undefined ? {} : { versionCommand }) };
      });
      return { ...parsed, profiles };
    },
    catch: () =>
      new MemoryError({
        code: "invalid",
        message:
          "Cannot load the explicit trusted extractor configuration; check its schema, unique profile IDs, placeholders and paths.",
      }),
  });
  return config;
});

export const resolveExtractorProfile = Effect.fn("resolveExtractorProfile")(function* (
  config: ExtractorConfig,
  id?: string,
) {
  const profile = config.profiles.find(
    (candidate) => candidate.id === (id ?? config.defaultProfile),
  );
  if (profile === undefined)
    return yield* new MemoryError({
      code: "invalid",
      message: "The requested extractor profile is not configured.",
    });
  return profile;
});

/** Hash options in stable key order; never put environment values or arguments into provenance. */
export function extractorProfileIdentity(profile: ExtractorProfile): string {
  if (profile.engine === "builtin") return builtinParserVersion;
  const options = {
    engine: profile.engine,
    command: profile.command,
    args: profile.args,
    env: Object.entries(profile.env ?? {}).sort(([left], [right]) => left.localeCompare(right)),
    timeoutMs: profile.timeoutMs ?? 60_000,
    outputMarkdown: profile.outputMarkdown ?? null,
    versionArgs: profile.versionArgs ?? null,
    versionCommand: profile.versionCommand ?? null,
    outputMode: profile.outputMode ?? "directory",
    mediaTypes: profile.mediaTypes ?? ["application/pdf"],
  };
  return `teamagent-extractor/1;${profile.engine};${profile.id};${profile.version};${digest(JSON.stringify(options))}`;
}

async function probeProfile(
  profile: ExternalExtractorProfile,
  directory: string,
  signal: AbortSignal,
): Promise<boolean> {
  if (isAbsolute(profile.command)) await access(profile.command, constants.X_OK);
  else if (
    Bun.which(profile.command, { PATH: profile.env?.PATH ?? process.env.PATH ?? "" }) === null
  )
    throw new Error("Extractor executable is not available");
  if (profile.versionArgs === undefined) return false;
  const result = await runExtractorProcess(
    profile.versionCommand ?? profile.command,
    profile.versionArgs,
    {
      signal,
      cwd: directory,
      env: { ...processEnvironment(), ...profile.env },
      bytes: 16_384,
      timeoutMs: Math.min(profile.timeoutMs ?? 60_000, 30_000),
    },
  );
  const actual = new TextDecoder("utf-8", { fatal: true }).decode(result).trim();
  const escapedVersion = profile.version.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  if (!new RegExp(`(^|[^a-zA-Z0-9._+-])${escapedVersion}($|[^a-zA-Z0-9._+-])`).test(actual))
    throw new Error("Extractor installed version does not match configured version");
  return true;
}

export const inspectExtractorProfile = Effect.fn("inspectExtractorProfile")(function* (
  profile: ExtractorProfile,
) {
  const decoded = yield* Schema.decodeUnknownEffect(ExtractorProfileSchema)(profile, {
    onExcessProperty: "error",
  }).pipe(
    Effect.mapError(
      () => new MemoryError({ code: "invalid", message: "Invalid extractor profile." }),
    ),
  );
  return yield* Effect.tryPromise({
    try: async (signal) => {
      profile = decoded;
      if (profile.engine === "builtin")
        return {
          available: true,
          profileId: profile.id,
          engine: profile.engine,
          expectedVersion: builtinParserVersion,
          observedVersion: null,
          verified: false,
          parserVersion: builtinParserVersion,
          message:
            "Builtin profile selected; memory doctor checks the pinned Python worker separately.",
        };
      try {
        validateProfile(profile);
        const verified = await probeProfile(profile, process.cwd(), signal);
        return {
          available: true,
          profileId: profile.id,
          engine: profile.engine,
          expectedVersion: profile.version,
          observedVersion: verified ? profile.version : null,
          verified,
          parserVersion: extractorProfileIdentity(profile),
          message: verified
            ? "Extractor executable and declared version verified."
            : "Extractor executable found; no version probe is configured.",
        };
      } catch {
        signal.throwIfAborted();
        return {
          available: false,
          profileId: profile.id,
          engine: profile.engine,
          expectedVersion: profile.version,
          observedVersion: null,
          verified: false,
          parserVersion: extractorProfileIdentity(profile),
          message:
            "Extractor unavailable, version probe failed, or installed version differs from configuration.",
        };
      }
    },
    catch: () =>
      new MemoryError({ code: "cancelled", message: "Extractor inspection was cancelled." }),
  });
});

interface OutputFile {
  readonly path: string;
  readonly absolute: string;
  readonly size: number;
}

async function outputFiles(directory: string, signal: AbortSignal): Promise<OutputFile[]> {
  const files: OutputFile[] = [];
  let bytes = 0;
  let entries = 0;
  async function visit(current: string, prefix: string, depth: number): Promise<void> {
    signal.throwIfAborted();
    if (depth > importLimits.depth) throw new Error("Extractor output nesting limit exceeded");
    for (const entry of await readdir(current, { withFileTypes: true })) {
      if (++entries > importLimits.entries)
        throw new Error("Extractor output entry limit exceeded");
      const path = prefix + entry.name;
      const absolute = join(current, entry.name);
      const info = await lstat(absolute);
      if (
        !safeRelative(path) ||
        info.isSymbolicLink() ||
        !contained(directory, await realpath(absolute))
      )
        throw new Error("Unsafe extractor output path");
      if (info.isDirectory()) await visit(absolute, `${path}/`, depth + 1);
      else if (info.isFile()) {
        bytes += info.size;
        if (
          info.size > importLimits.fileBytes ||
          bytes > importLimits.totalBytes ||
          files.length >= importLimits.files
        )
          throw new Error("Extractor output size or file limit exceeded");
        files.push({ path, absolute, size: info.size });
      } else throw new Error("Extractor output is not a regular file");
    }
  }
  await visit(directory, "", 0);
  return files.sort((left, right) => left.path.localeCompare(right.path));
}

async function readOutput(file: OutputFile): Promise<Uint8Array> {
  const handle = await open(file.absolute, constants.O_RDONLY | (constants.O_NOFOLLOW ?? 0));
  try {
    const info = await handle.stat();
    if (!info.isFile() || info.size !== file.size || info.size > importLimits.fileBytes)
      throw new Error("Extractor output changed after validation");
    const bytes = Buffer.alloc(info.size);
    let offset = 0;
    while (offset < bytes.length) {
      const read = await handle.read(bytes, offset, bytes.length - offset, offset);
      if (read.bytesRead === 0) throw new Error("Incomplete extractor output");
      offset += read.bytesRead;
    }
    if ((await handle.stat()).size !== info.size)
      throw new Error("Extractor output changed while reading");
    return bytes;
  } finally {
    await handle.close();
  }
}

function outputPathKey(path: string): string {
  const absolute = resolve(path);
  return process.platform === "win32" ? absolute.toLowerCase() : absolute;
}

function replaceImageReferences(
  markdown: string,
  markdownPath: string,
  outputDirectory: string,
  files: readonly OutputFile[],
  assets: ReadonlyMap<string, string>,
): string {
  const validatedFiles = new Set(files.map((file) => outputPathKey(file.absolute)));
  function asset(source: string): string | undefined {
    let path: string;
    try {
      if (/^file:/i.test(source)) {
        // File URLs are rebased only when they identify an already validated output asset.
        return assets.get(outputPathKey(fileURLToPath(new URL(source))));
      }
      path = decodeURIComponent(source);
    } catch {
      return undefined;
    }
    if (isAbsolute(path)) return assets.get(outputPathKey(path));
    if (/^[a-z][a-z0-9+.-]*:/i.test(path)) return undefined;
    const local = outputPathKey(resolve(dirname(markdownPath), path));
    // An existing Markdown-relative file wins, even when it is not an imported image.
    if (validatedFiles.has(local)) return assets.get(local);
    return assets.get(outputPathKey(resolve(outputDirectory, path)));
  }
  function replaceTarget(source: string): string {
    const angled = source.startsWith("<") && source.endsWith(">");
    const target = asset(angled ? source.slice(1, -1) : source);
    return target === undefined ? source : angled ? `<${target}>` : target;
  }
  // One pass over source references prevents a newly generated asset URL from being
  // interpreted again as a different extractor output path.
  return markdown.replace(
    /(\]\([ \t]*)(<[^>\r\n]+>|(?:[^\s()]|\([^\s()]*\))+)(?=[ \t)]|$)|(^[ \t]*\[[^\]\r\n]+\]:[ \t]*)(<[^>\r\n]+>|[^ \t\r\n]+)|(\bsrc=)(["'])([^"'\r\n]+)\6/gim,
    (
      match: string,
      inlinePrefix: string | undefined,
      inlineUrl: string | undefined,
      referencePrefix: string | undefined,
      referenceUrl: string | undefined,
      htmlPrefix: string | undefined,
      quote: string | undefined,
      htmlUrl: string | undefined,
    ) => {
      if (inlinePrefix !== undefined && inlineUrl !== undefined)
        return inlinePrefix + replaceTarget(inlineUrl);
      if (referencePrefix !== undefined && referenceUrl !== undefined)
        return referencePrefix + replaceTarget(referenceUrl);
      if (htmlPrefix !== undefined && quote !== undefined && htmlUrl !== undefined)
        return htmlPrefix + quote + replaceTarget(htmlUrl) + quote;
      return match;
    },
  );
}

export async function convertExternalDocument(options: {
  readonly bytes: Uint8Array;
  readonly name: string;
  readonly directory: string;
  readonly mediaType: string;
  readonly signal: AbortSignal;
  readonly extractor: ExternalExtractorProfile;
}) {
  const profile = Schema.decodeUnknownSync(ExtractorProfileSchema)(options.extractor, {
    onExcessProperty: "error",
  });
  validateProfile(profile);
  if (profile.engine === "builtin") throw new Error("Expected external extractor profile");
  options.signal.throwIfAborted();
  if (options.bytes.byteLength > importLimits.fileBytes)
    throw new Error("Document exceeds import limit");
  await mkdir(options.directory, { recursive: true });
  const base = await realpath(options.directory);
  const directory = await mkdtemp(join(base, "extractor-"));
  if (!contained(base, directory) || directory === base)
    throw new Error("Invalid extractor staging directory");
  const env = { ...processEnvironment(), ...profile.env };
  const warnings: string[] = [];
  try {
    if (!(await probeProfile(profile, directory, options.signal)))
      warnings.push(
        "Extractor version is declared in trusted configuration; no version probe was configured.",
      );
    const input = join(directory, `input${documentExtension(options.name, options.mediaType)}`);
    const outputDir = join(directory, "output");
    await mkdir(outputDir);
    await writeFile(input, options.bytes, { flag: "wx", mode: 0o600 });
    const args = profile.args.map((arg) =>
      arg.replaceAll("{input}", input).replaceAll("{outputDir}", outputDir),
    );
    const stdout = await runExtractorProcess(profile.command, args, {
      signal: options.signal,
      cwd: directory,
      env,
      bytes: profile.outputMode === "stdout" ? importLimits.fileBytes : 1_048_576,
      timeoutMs: profile.timeoutMs ?? 60_000,
    });
    if (profile.outputMode === "stdout") {
      const path = join(outputDir, profile.outputMarkdown ?? "document.md");
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, stdout, { flag: "wx", mode: 0o600 });
    }
    const files = await outputFiles(outputDir, options.signal);
    const candidates = files.filter((file) =>
      profile.outputMarkdown === undefined
        ? /\.(md|markdown)$/i.test(file.path)
        : file.path === profile.outputMarkdown.replaceAll("\\", "/"),
    );
    if (candidates.length !== 1 || candidates[0] === undefined)
      throw new Error("Extractor must produce exactly one selected Markdown document");
    const selected = candidates[0];
    let markdown = new TextDecoder("utf-8", { fatal: true }).decode(await readOutput(selected));
    if (markdown.includes("\0")) throw new Error("Extractor produced binary Markdown");
    const images: { name: string; mediaType: string; base64: string }[] = [];
    const seen = new Set<string>();
    const assets = new Map<string, string>();
    for (const file of files) {
      if (!/\.(?:png|jpe?g|gif|webp)$/i.test(file.path)) continue;
      const bytes = await readOutput(file);
      const mediaType = sourceMediaType("", bytes);
      const extensions: Readonly<Record<string, string>> = {
        "image/png": "png",
        "image/jpeg": "jpg",
        "image/gif": "gif",
        "image/webp": "webp",
      };
      const imageExtension = extensions[mediaType];
      if (imageExtension === undefined)
        throw new Error("Extractor image content does not match a supported raster type");
      const name = `${digest(bytes)}.${imageExtension}`;
      if (!seen.has(name)) {
        if (images.length >= 128) throw new Error("Extractor image count limit exceeded");
        images.push({ name, mediaType, base64: Buffer.from(bytes).toString("base64") });
        seen.add(name);
      }
      assets.set(outputPathKey(file.absolute), `assets/${name}`);
    }
    markdown = replaceImageReferences(markdown, selected.absolute, outputDir, files, assets);
    const ignored = files.filter(
      (file) => file !== selected && !/\.(?:png|jpe?g|gif|webp)$/i.test(file.path),
    );
    if (ignored.length > 0)
      warnings.push(
        `Extractor produced ${ignored.length} sidecar files that were not imported; only selected Markdown and raster images are retained.`,
      );
    if (
      /!\[[^\]]*\]\(\s*(?:https?:|data:|file:|\/|[a-z]:\\)/i.test(markdown) ||
      /<img\b/i.test(markdown)
    )
      warnings.push(
        "External or embedded image references are preserved as text and are not fetched; only contained raster files are imported.",
      );
    if (/!\[[^\]]*\]\((?!<?assets\/[a-f0-9]{64}\.(?:png|jpg|gif|webp)(?:>|[\s)]))/.test(markdown))
      warnings.push(
        "Some image references do not resolve to imported raster assets; their source text is retained without fetching or substituting images.",
      );
    if (Buffer.byteLength(markdown) > importLimits.fileBytes)
      throw new Error("Normalized extractor Markdown exceeds import limit");
    return {
      markdown,
      images,
      links: [],
      warnings,
      parserVersion: extractorProfileIdentity(profile),
    };
  } finally {
    await removeExtractorStaging(directory, base);
  }
}
